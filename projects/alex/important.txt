How to add data in knowledge base
1) 
URL="https://wcxdfv6cqxj3vtcx62mojcoceq0ssgiq.lambda-url.us-east-2.on.aws"

# Specific topic
curl -sS --max-time 630 -X POST "$URL/research" \
  -H "Content-Type: application/json" \
  -d '{"topic":"US large-cap market outlook and interest rates"}'

# Let the agent choose a trending topic
curl -sS --max-time 630 -X POST "$URL/research" \
  -H "Content-Type: application/json" \
  -d '{}'

# Dedicated auto endpoint (GET, used by the scheduler)
curl -sS --max-time 630 "$URL/research/auto"

2) 
cd backend/researcher

# Let the agent pick a trending topic
uv run test_research.py

# Or give it a specific topic
uv run test_research.py "AAPL earnings and AI strategy outlook"

How to check data in knowledge base
keys=$(aws s3vectors list-vectors \
  --vector-bucket-name alex-vectors-304896042944 \
  --index-name financial-research --max-results 500 \
  --region us-east-2 --query 'vectors[].key' --output text)

aws s3vectors get-vectors \
  --vector-bucket-name alex-vectors-304896042944 \
  --index-name financial-research --keys $keys \
  --return-metadata --region us-east-2 \
  --query 'vectors[].metadata.[timestamp,topic]' --output text | sort
